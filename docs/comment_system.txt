4. CONTROLLER FUNCTIONS
---------------------
```javascript
// controllers/commentController.js

// Create a new comment
const createComment = async (req, res) => {
    try {
        const { content, media } = req.body;
        const comment = await commentService.createComment({
            content,
            postId: req.params.postId,
            author: req.user._id,
            media
        });
        res.status(201).json(comment);
    } catch (error) {
        res.status(500).json({ error: error.message });
    }
};

// Get comments for a post
const getComments = async (req, res) => {
    try {
        const comments = await commentService.getCommentsByPost(req.params.postId);
        res.json(comments);
    } catch (error) {
        res.status(500).json({ error: error.message });
    }
};

// Update a comment
const updateComment = async (req, res) => {
    try {
        const comment = await commentService.updateComment(
            req.params.commentId,
            req.body,
            req.user._id
        );
        res.json(comment);
    } catch (error) {
        res.status(500).json({ error: error.message });
    }
};

// Delete a comment
const deleteComment = async (req, res) => {
    try {
        await commentService.deleteComment(req.params.commentId, req.user._id);
        res.status(204).send();
    } catch (error) {
        res.status(500).json({ error: error.message });
    }
};
```

5. SERVICE FUNCTIONS
------------------
```javascript
// services/commentService.js

// Create comment
const createComment = async ({ content, postId, author, media, parentComment }) => {
    const comment = new Comment({
        content,
        postId,
        author,
        media,
        parentComment
    });
    
    await comment.save();
    
    // Update post's comment count
    await Post.findByIdAndUpdate(postId, {
        $inc: { commentCount: 1 }
    });
    
    return comment;
};

// Get comments for a post
const getCommentsByPost = async (postId) => {
    return await Comment.find({ postId })
        .populate('author', 'username profilePicture')
        .populate('likes', 'username')
        .sort({ createdAt: -1 });
};

// Update comment
const updateComment = async (commentId, updates, userId) => {
    const comment = await Comment.findOne({
        _id: commentId,
        author: userId
    });
    
    if (!comment) {
        throw new Error('Comment not found or unauthorized');
    }
    
    Object.assign(comment, updates);
    comment.updatedAt = new Date();
    await comment.save();
    
    return comment;
};
```

```jsx
// components/Comment/CommentForm.jsx
const CommentForm = ({ postId, parentComment, onComment }) => {
    const [content, setContent] = useState('');
    const [media, setMedia] = useState([]);
    
    const handleSubmit = async (e) => {
        e.preventDefault();
        try {
            const response = await api.post(`/posts/${postId}/comments`, {
                content,
                media,
                parentComment
            });
            onComment(response.data);
            setContent('');
            setMedia([]);
        } catch (error) {
            console.error('Error creating comment:', error);
        }
    };
    
    return (
        <form onSubmit={handleSubmit} className={styles.form}>
            <textarea
                value={content}
                onChange={(e) => setContent(e.target.value)}
                placeholder="Write a comment..."
                maxLength={280}
            />
            <MediaUploader onMediaSelect={setMedia} />
            <button type="submit" disabled={!content.trim()}>
                Comment
            </button>
        </form>
    );
};

// components/Comment/CommentList.jsx
const CommentList = ({ postId }) => {
    const [comments, setComments] = useState([]);
    const [loading, setLoading] = useState(true);
    
    useEffect(() => {
        const fetchComments = async () => {
            try {
                const response = await api.get(`/posts/${postId}/comments`);
                setComments(response.data);
            } catch (error) {
                console.error('Error fetching comments:', error);
            } finally {
                setLoading(false);
            }
        };
        
        fetchComments();
    }, [postId]);
    
    return (
        <div className={styles.commentList}>
            {loading ? (
                <LoadingSpinner />
            ) : (
                comments.map(comment => (
                    <CommentCard
                        key={comment._id}
                        comment={comment}
                        onUpdate={setComments}
                    />
                ))
            )}
        </div>
    );
};
```

7. SOCKET.IO EVENTS
-----------------
```javascript
// Socket events for real-time comments
socket.on('newComment', (comment) => {
    // Add new comment to the list
    setComments(prev => [comment, ...prev]);
});

socket.on('commentUpdated', (updatedComment) => {
    // Update existing comment
    setComments(prev => prev.map(comment => 
        comment._id === updatedComment._id ? updatedComment : comment
    ));
});

socket.on('commentDeleted', (commentId) => {
    // Remove deleted comment
    setComments(prev => prev.filter(comment => comment._id !== commentId));
});
```

9. ERROR HANDLING
---------------
```javascript
// Error handling middleware
const handleCommentError = (error, req, res, next) => {
    if (error.name === 'ValidationError') {
        return res.status(400).json({
            error: 'Validation Error',
            details: error.errors
        });
    }
    
    if (error.name === 'UnauthorizedError') {
        return res.status(403).json({
            error: 'Unauthorized',
            message: 'You do not have permission to perform this action'
        });
    }
    
    res.status(500).json({
        error: 'Server Error',
        message: error.message
    });
};
```

        );

    });
    
    test('should delete a comment', async () => {

    });
