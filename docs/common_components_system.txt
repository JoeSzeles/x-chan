COMMON COMPONENTS SYSTEM
=======================

1. EXISTING COMMON COMPONENTS
---------------------------
1.1 LoadingSpinner
    - Location: components/LoadingSpinner.jsx
    - Props:
        * size: "sm" | "md" | "lg"
        * color: string (default: "indigo-600")
    - Features:
        * Responsive sizing
        * Customizable colors
        * Smooth animation
    - Usage:
        ```javascript
        <LoadingSpinner size="sm" color="blue-500" />
        ```

1.2 ErrorMessage
    - Location: components/ErrorMessage.jsx
    - Props:
        * message: string
        * onRetry: function (optional)
    - Features:
        * Error message display
        * Retry button
        * Error icon
    - Usage:
        ```javascript
        <ErrorMessage 
            message="Failed to load data" 
            onRetry={handleRetry} 
        />
        ```

1.3 PostCard
    - Location: components/PostCard.jsx
    - Props:
        * post: object
        * onLike: function
        * onComment: function
        * onRepost: function
    - Features:
        * Post content display
        * Interaction buttons
        * User info
        * Timestamp
    - Usage:
        ```javascript
        <PostCard 
            post={postData} 
            onLike={handleLike}
            onComment={handleComment}
            onRepost={handleRepost}
        />
        ```

    ```javascript
        // 4. Define handlers

            setIsLoading(true);
            try {
                await onAction();
            } catch (error) {
                console.error(error);
            } finally {
                setIsLoading(false);
            }
        };

        // 5. Define effects if needed
        useEffect(() => {
            // Effect logic
        }, [dependencies]);

        return (
            <motion.div
                className={`component-base ${className}`}
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                exit={{ opacity: 0 }}
                {...props}
            >
                {/* Component content */}
                {isLoading ? (
                    <LoadingSpinner size="sm" />
                ) : (
                    <button
                        onClick={handleAction}
                        className="action-button"
                        disabled={isLoading}
                    >
                        Action
                    </button>
                )}
            </motion.div>
        );
    };

    // 6. Define prop types

        prop1: PropTypes.string.isRequired,
    ```

4. EXAMPLE: ADDING A NEW "USER CARD" COMPONENT
--------------------------------------------
4.1 Component Implementation
    ```javascript
    import PropTypes from "prop-types";
    import { useState } from "react";
    import { motion } from "framer-motion";
    import { FaUser, FaEllipsisH } from "react-icons/fa";

    const UserCard = ({
        user,
        onFollow,
        onMessage,
        showActions = true,
        className = "",
    }) => {
        const [isFollowing, setIsFollowing] = useState(user.isFollowing);
        const [isLoading, setIsLoading] = useState(false);

        const handleFollow = async () => {
            setIsLoading(true);
            try {
                await onFollow(user._id);
                setIsFollowing(!isFollowing);
            } catch (error) {
                console.error(error);
            } finally {
                setIsLoading(false);
            }
        };

        return (
            <motion.div
                className={`bg-white rounded-lg shadow p-4 ${className}`}
                initial={{ opacity: 0 }}
                animate={{ opacity: 1 }}
                exit={{ opacity: 0 }}
            >
                <div className="flex items-center space-x-4">
                    <div className="w-12 h-12 rounded-full bg-gray-200 flex items-center justify-center">
                        {user.profilePicture ? (
                            <img
                                src={user.profilePicture}
                                alt={user.username}
                                className="w-full h-full rounded-full object-cover"
                            />
                        ) : (
                            <FaUser className="text-gray-400 text-xl" />
                        )}
                    </div>
                    <div className="flex-1">
                        <h3 className="font-semibold">{user.fullName}</h3>
                        <p className="text-gray-500">@{user.username}</p>
                    </div>
                    {showActions && (
                        <div className="flex space-x-2">
                            <button
                                onClick={handleFollow}
                                disabled={isLoading}
                                className={`px-3 py-1 rounded-full text-sm ${
                                    isFollowing
                                        ? "bg-gray-200 text-gray-700"
                                        : "bg-indigo-600 text-white"
                                }`}
                            >
                                {isLoading ? (
                                    <LoadingSpinner size="sm" />
                                ) : isFollowing ? (
                                    "Following"
                                ) : (
                                    "Follow"
                                )}
                            </button>
                            <button
                                onClick={() => onMessage(user._id)}
                                className="p-2 rounded-full hover:bg-gray-100"
                            >
                                <FaEllipsisH />
                            </button>
                        </div>
                    )}
                </div>
            </motion.div>
        );
    };

    UserCard.propTypes = {
        user: PropTypes.shape({
            _id: PropTypes.string.isRequired,
            username: PropTypes.string.isRequired,
            fullName: PropTypes.string.isRequired,
            profilePicture: PropTypes.string,
            isFollowing: PropTypes.bool,
        }).isRequired,
        onFollow: PropTypes.func.isRequired,
        onMessage: PropTypes.func.isRequired,
        showActions: PropTypes.bool,
        className: PropTypes.string,
    };

    export default UserCard;
    ```

    ```javascript

            <div className="space-y-4">

                        key={user._id}

        );
    };
    ```
