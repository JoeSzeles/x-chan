2. TEMPLATE FOR NEW FORM COMPONENTS
---------------------------------
2.1 Frontend Implementation Template
    ```javascript
    // 1. Create new form component
    import { useState } from "react";
    import { useMutation } from "@tanstack/react-query";
    import { toast } from "react-hot-toast";
    import LoadingSpinner from "./LoadingSpinner";

    const NewForm = () => {
        // 2. Define form state
        const [formData, setFormData] = useState({
            field1: "",
            field2: "",
            file: null,
        });
        const [isUploading, setIsUploading] = useState(false);
        const [preview, setPreview] = useState(null);

        // 3. Handle form input changes
        const handleInputChange = (e) => {
            const { name, value } = e.target;
            setFormData({ ...formData, [name]: value });
        };

        // 4. Handle file upload
        const handleFileChange = (e) => {
            const file = e.target.files[0];
            if (file) {
                setFormData({ ...formData, file });
                const reader = new FileReader();
                reader.onloadend = () => {
                    setPreview(reader.result);
                };
                reader.readAsDataURL(file);
            }
        };

        // 5. Add mutation
        const { mutate: submitForm, isPending } = useMutation({
            mutationFn: async () => {
                try {
                    const formDataToSend = new FormData();
                    Object.keys(formData).forEach((key) => {
                        formDataToSend.append(key, formData[key]);
                    });

                    const res = await fetch("/api/endpoint", {
                        method: "POST",
                        body: formDataToSend,
                    });
                    const data = await res.json();
                    if (!res.ok) throw new Error(data.error);
                    return data;
                } catch (error) {
                    throw new Error(error);
                }
            },
            onSuccess: (data) => {
                toast.success("Form submitted successfully");
                // Handle success (e.g., redirect, update state)
            },
            onError: (error) => {
                toast.error(error.message);
            },
        });

        // 6. Handle form submission
        const handleSubmit = (e) => {
            e.preventDefault();
            submitForm();
        };

        return (
            <form onSubmit={handleSubmit} className="space-y-4">
                {/* Text Input */}
                <div>
                    <label className="block text-sm font-medium text-gray-700">
                        Field 1
                    </label>
                    <input
                        type="text"
                        name="field1"
                        value={formData.field1}
                        onChange={handleInputChange}
                        className="mt-1 block w-full rounded-md border-gray-300 shadow-sm focus:border-indigo-500 focus:ring-indigo-500"
                    />
                </div>

                {/* File Input */}
                <div>
                    <label className="block text-sm font-medium text-gray-700">
                        File Upload
                    </label>
                    <input
                        type="file"
                        onChange={handleFileChange}
                        className="mt-1 block w-full"
                    />
                    {preview && (
                        <img
                            src={preview}
                            alt="Preview"
                            className="mt-2 h-32 w-32 object-cover"
                        />
                    )}
                </div>

                {/* Submit Button */}
                <button
                    type="submit"
                    disabled={isPending}
                    className="flex w-full justify-center rounded-md bg-indigo-600 px-3 py-2 text-sm font-semibold text-white shadow-sm hover:bg-indigo-500 focus-visible:outline focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-indigo-600"
                >
                    {isPending ? <LoadingSpinner size="sm" /> : "Submit"}
                </button>
            </form>
        );
    };

    export default NewForm;
    ```

4. EXAMPLE: ADDING A NEW "CREATE POLL" FORM
-----------------------------------------
4.1 Frontend Implementation
    ```javascript
    const CreatePoll = () => {
        const [formData, setFormData] = useState({
            question: "",
            options: ["", ""],
            duration: 24, // hours
        });

        const handleOptionChange = (index, value) => {
            const newOptions = [...formData.options];
            newOptions[index] = value;
            setFormData({ ...formData, options: newOptions });
        };

        const addOption = () => {
            setFormData({
                ...formData,
                options: [...formData.options, ""],
            });
        };

        const { mutate: createPoll, isPending } = useMutation({
            mutationFn: async () => {
                const res = await fetch("/api/polls", {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json",
                    },
                    body: JSON.stringify(formData),
                });
                const data = await res.json();
                if (!res.ok) throw new Error(data.error);
                return data;
            },
            onSuccess: () => {
                toast.success("Poll created successfully");
                // Handle success
            },
        });

        return (
            <form onSubmit={(e) => {
                e.preventDefault();
                createPoll();
            }}>
                <input
                    type="text"
                    value={formData.question}
                    onChange={(e) => setFormData({ ...formData, question: e.target.value })}
                    placeholder="Enter your question"
                />
                {formData.options.map((option, index) => (
                    <input
                        key={index}
                        type="text"
                        value={option}
                        onChange={(e) => handleOptionChange(index, e.target.value)}
                        placeholder={`Option ${index + 1}`}
                    />
                ))}
                <button type="button" onClick={addOption}>
                    Add Option
                </button>
                <button type="submit" disabled={isPending}>
                    {isPending ? <LoadingSpinner /> : "Create Poll"}
                </button>
            </form>
        );
    };
    ```

4.2 Backend Implementation
    ```javascript
    // In poll.routes.js
    router.post("/", protectRoute, createPoll);

    // In poll.controller.js
    export const createPoll = async (req, res, next) => {
        try {
            const { question, options, duration } = req.body;
            const userId = req.user._id;

            // Validate input
            if (!question || options.length < 2) {
                return next(errorHandler(400, "Invalid poll data"));
            }

            // Create poll
            const poll = await Poll.create({
                question,
                options: options.map(option => ({
                    text: option,
                    votes: 0
                })),
                duration,
                createdBy: userId,
                expiresAt: new Date(Date.now() + duration * 60 * 60 * 1000)
            });

            res.status(201).json(poll);
        } catch (error) {
            next(error);
        }
    };
    ```
