2.2 Backend Implementation Template
    ```javascript
    // 1. Create new route file
    import express from "express";
    import { protectRoute } from "../middleware/auth.js";
    import {
        getPageData,
        performAction,
    } from "../controllers/pageController.js";

    const router = express.Router();

    // 2. Define routes
    router.get("/", protectRoute, getPageData);
    router.post("/action", protectRoute, performAction);

    export default router;

    // 3. Create controller
    export const getPageData = async (req, res, next) => {
        try {
            const userId = req.user._id;
            
            // Fetch page data
            const pageData = await fetchPageData(userId);

            res.status(200).json(pageData);
        } catch (error) {
            next(error);
        }
    };

    export const performAction = async (req, res, next) => {
        try {
            const { actionData } = req.body;
            const userId = req.user._id;

            // Validate input
            if (!actionData) {
                return next(errorHandler(400, "Action data is required"));
            }

            // Perform action
            const result = await processAction(userId, actionData);

            res.status(200).json({
                message: "Action completed successfully",
                data: result,
            });
        } catch (error) {
            next(error);
        }
    };
    ```

4. EXAMPLE: ADDING A NEW "EXPLORE" PAGE
-------------------------------------
4.1 Frontend Implementation
    ```javascript
    const Explore = () => {
        const [searchQuery, setSearchQuery] = useState("");
        const [selectedCategory, setSelectedCategory] = useState("all");

        const { data: exploreData, isLoading } = useQuery({
            queryKey: ["explore", searchQuery, selectedCategory],
            queryFn: async () => {
                const res = await fetch(
                    `/api/explore?q=${searchQuery}&category=${selectedCategory}`
                );
                const data = await res.json();
                if (!res.ok) throw new Error(data.error);
                return data;
            },
        });

        return (
            <div className="container mx-auto px-4 py-8">
                {/* Search and Filter */}
                <div className="mb-8">
                    <input
                        type="text"
                        value={searchQuery}
                        onChange={(e) => setSearchQuery(e.target.value)}
                        placeholder="Search..."
                        className="w-full p-2 border rounded-md"
                    />
                    <select
                        value={selectedCategory}
                        onChange={(e) => setSelectedCategory(e.target.value)}
                        className="mt-4 p-2 border rounded-md"
                    >
                        <option value="all">All Categories</option>
                        <option value="trending">Trending</option>
                        <option value="news">News</option>
                        <option value="sports">Sports</option>
                    </select>
                </div>

                {/* Content Grid */}
                <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-6">
                    {exploreData?.items.map((item) => (
                        <ExploreCard key={item._id} item={item} />
                    ))}
                </div>
            </div>
        );
    };
    ```

4.2 Backend Implementation
    ```javascript
    // In explore.routes.js

    // In explore.controller.js
    export const getExploreData = async (req, res, next) => {
    ```
