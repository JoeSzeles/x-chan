4.1 Frontend Changes

4.2 Backend Changes
    ```javascript
    // In post.routes.js
    router.post("/save/:id", protectRoute, savePost);

    // In post.controller.js
    export const savePost = async (req, res, next) => {
        try {
            const { id } = req.params;
            const userId = req.user._id;

            const post = await Post.findById(id);
            if (!post) return next(errorHandler(404, "Post not found"));

            const isSaved = post.savedBy.includes(userId);
            
            if (isSaved) {
                await Post.findByIdAndUpdate(id, {
                    $pull: { savedBy: userId }
                });
            } else {
                await Post.findByIdAndUpdate(id, {
                    $push: { savedBy: userId }
                });
            }

            res.status(200).json({
                isSaved: !isSaved,
                savedBy: isSaved 
                    ? post.savedBy.filter(id => id.toString() !== userId.toString())
                    : [...post.savedBy, userId]
            });
        } catch (error) {
            next(error);
        }
    };
    ```
