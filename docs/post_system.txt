POST CARD SYSTEM
===============

1. BASE POST STRUCTURE
--------------------
1.1 Post Model
    ```javascript
    // post.model.js
    const postSchema = new mongoose.Schema({
        author: {
            type: mongoose.Schema.Types.ObjectId,
            ref: "User",
            required: true,
        },
        type: {
            type: String,
            enum: ["standard", "service", "portfolio", "availability", "credentials"],
            required: true,
        },
        content: {
            text: String,
            media: [{
                type: String,
                url: String,
                alt: String,
                order: Number,
            }],
        },
        metadata: {
            likes: { type: Number, default: 0 },
            comments: { type: Number, default: 0 },
            shares: { type: Number, default: 0 },
            views: { type: Number, default: 0 },
        },
        visibility: {
            type: String,
            enum: ["public", "followers", "private"],
            default: "public",
        },
        tags: [String],
        createdAt: {
            type: Date,
            default: Date.now,
        },
        updatedAt: {
            type: Date,
            default: Date.now,
        },
    });

    // Add indexes
    postSchema.index({ author: 1, createdAt: -1 });
    postSchema.index({ type: 1, createdAt: -1 });
    postSchema.index({ tags: 1 });
    ```

1.2 Base Post Component
    ```javascript
    // components/Post/BasePost.jsx
    const BasePost = ({ post, children }) => {
        return (
            <div className="post-card">
                {/* Post Header */}
                <div className="post-header">
                    <UserAvatar user={post.author} />
                    <div className="post-meta">
                        <UserName user={post.author} />
                        <PostTimestamp date={post.createdAt} />
                    </div>
                    <PostActions post={post} />
                </div>

                {/* Post Content */}
                <div className="post-content">
                    {children}
                </div>

                {/* Post Footer */}
                <div className="post-footer">
                    <PostInteractions post={post} />
                    <PostStats post={post} />
                </div>
            </div>
        );
    };
    ```

4. POST FACTORY
-------------
4.1 Post Component Factory
    ```javascript
    // components/Post/PostFactory.jsx
    const PostFactory = ({ post }) => {
        const components = {
            standard: StandardPost,
            service: ServicePost,
            portfolio: PortfolioPost,
            availability: AvailabilityPost,
            credentials: CredentialsPost,
        };

        const PostComponent = components[post.type] || StandardPost;
        return <PostComponent post={post} />;
    };
    ```

4.2 Editor Factory
    ```javascript
    // components/Editor/EditorFactory.jsx
    const EditorFactory = ({ post, onSave }) => {
        const editors = {
            standard: StandardEditor,
            service: ServiceEditor,
            portfolio: PortfolioEditor,
            availability: AvailabilityEditor,
            credentials: CredentialsEditor,
        };

        const EditorComponent = editors[post.type] || StandardEditor;
        return <EditorComponent post={post} onSave={onSave} />;
    };
    ```
