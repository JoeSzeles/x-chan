    // 1. Create feature component
    import { useState } from "react";

    import { toast } from "react-hot-toast";
    import { motion } from "framer-motion";

    const NewFeature = ({ targetId, currentUser }) => {
        const [isActive, setIsActive] = useState(false);
        const [isLoading, setIsLoading] = useState(false);
        const queryClient = useQueryClient();

        // 2. Add mutation
        const { mutate: performAction } = useMutation({
            mutationFn: async () => {

                    method: "POST",
                });
                const data = await res.json();
                if (!res.ok) throw new Error(data.error);
                return data;
            },
            onSuccess: (data) => {
                setIsActive(data.isActive);
                // Update relevant queries
                queryClient.invalidateQueries(["relevantData"]);
                toast.success(data.message);
            },
            onError: (error) => {
                toast.error(error.message);
            },
        });

        // 3. Handle feature action
        const handleAction = () => {
            performAction();
        };

        return (
            <motion.button
                whileHover={{ scale: 1.05 }}
                whileTap={{ scale: 0.95 }}
                onClick={handleAction}
                disabled={isLoading}
                className={`feature-button ${isActive ? "active" : ""}`}
            >
                {isLoading ? (
                    <LoadingSpinner size="sm" />
                ) : (
                    <FeatureIcon isActive={isActive} />
                )}
                <span className="count">{count}</span>

        try {
            const { id } = req.params;
            const userId = req.user._id;

            // Validate input
            const target = await TargetModel.findById(id);
            if (!target) {
                return next(errorHandler(404, "Target not found"));
            }

            // Check if action already exists
            const existingAction = await FeatureModel.findOne({
                user: userId,
                target: id,
            });

            let result;
            if (existingAction) {
                // Remove action
                await existingAction.remove();
                result = { isActive: false };
            } else {
                // Create new action
                const newAction = await FeatureModel.create({
                    user: userId,
                    target: id,
                });
                result = { isActive: true };
            }

            // Update counts
            await updateFeatureCounts(id);

                await createNotification({

                    target: id,

            }

            res.status(200).json(result);
        } catch (error) {
            next(error);
        }
    };

4. EXAMPLE: ADDING A NEW "REACT" FEATURE
-------------------------------------
4.1 Frontend Implementation
    ```javascript
    const ReactButton = ({ postId, currentUser }) => {
        const [selectedReaction, setSelectedReaction] = useState(null);
        const [showPicker, setShowPicker] = useState(false);
        const queryClient = useQueryClient();

        const { mutate: addReaction } = useMutation({
            mutationFn: async (reaction) => {
                const res = await fetch(`/api/posts/${postId}/react`, {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json",
                    },
                    body: JSON.stringify({ reaction }),
                });
                const data = await res.json();
                if (!res.ok) throw new Error(data.error);
                return data;
            },
            onSuccess: (data) => {
                setSelectedReaction(data.reaction);
                queryClient.invalidateQueries(["post", postId]);
                toast.success("Reaction added!");
            },
        });

        return (
            <div className="relative">
                <button
                    onClick={() => setShowPicker(!showPicker)}
                    className="flex items-center space-x-1 p-2 hover:bg-gray-100 rounded-full"
                >
                    {selectedReaction ? (
                        <span className="text-2xl">{selectedReaction}</span>
                    ) : (
                        <FaRegSmile className="text-gray-500" />
                    )}
                </button>

                {showPicker && (
                    <div className="absolute bottom-full left-0 mb-2 bg-white rounded-lg shadow-lg p-2">
                        <div className="grid grid-cols-6 gap-2">
                            {["😀", "❤️", "😂", "😮", "😢", "😡"].map((emoji) => (
                                <button
                                    key={emoji}
                                    onClick={() => {
                                        addReaction(emoji);
                                        setShowPicker(false);
                                    }}
                                    className="text-2xl hover:scale-110 transition-transform"
                                >
                                    {emoji}
                                </button>
                            ))}
                        </div>
                    </div>
                )}
            </div>
        );
    };
    ```

4.2 Backend Implementation
    ```javascript
    // In post.routes.js
    router.post("/:id/react", protectRoute, addReaction);

    // In post.controller.js
    export const addReaction = async (req, res, next) => {
        try {
            const { id } = req.params;
            const { reaction } = req.body;
            const userId = req.user._id;

            // Validate reaction
            const validReactions = ["😀", "❤️", "😂", "😮", "😢", "😡"];
            if (!validReactions.includes(reaction)) {
                return next(errorHandler(400, "Invalid reaction"));
            }

            // Update or create reaction
            const existingReaction = await Reaction.findOne({
                user: userId,
                post: id,
            });

            if (existingReaction) {
                existingReaction.reaction = reaction;
                await existingReaction.save();
            } else {
                await Reaction.create({
                    user: userId,
                    post: id,
                    reaction,
                });
            }

            // Update post reaction counts
            await updateReactionCounts(id);

            // Create notification
            const post = await Post.findById(id);
            if (post.user.toString() !== userId.toString()) {
                await createNotification({
                    type: "REACTION",
                    sender: userId,
                    receiver: post.user,
                    post: id,
                    reaction,
                });
            }

            res.status(200).json({
                reaction,
                message: "Reaction added successfully",
            });
        } catch (error) {
            next(error);
        }
    };
    ```

4.3 Model Updates
    ```javascript
    // In reaction.model.js
    const reactionSchema = new mongoose.Schema({
        user: {
            type: mongoose.Schema.Types.ObjectId,
            ref: "User",
            required: true,
        },
        post: {
            type: mongoose.Schema.Types.ObjectId,
            ref: "Post",
            required: true,
        },
        reaction: {
            type: String,
            required: true,
            enum: ["😀", "❤️", "😂", "😮", "😢", "😡"],
        },
        createdAt: {
            type: Date,
            default: Date.now,
        },
    });

    // Add indexes
    reactionSchema.index({ user: 1, post: 1 }, { unique: true });
    reactionSchema.index({ post: 1, createdAt: -1 });

    // Add methods
    reactionSchema.statics.getReactionCounts = async function(postId) {
        return this.aggregate([
            { $match: { post: postId } },
            { $group: { _id: "$reaction", count: { $sum: 1 } } },
        ]);
    };

    export const Reaction = mongoose.model("Reaction", reactionSchema);
    ```
