    ```javascript
    // list.model.js

        name: {
            type: String,

            type: String,

        },
        createdAt: {
            type: Date,
            default: Date.now,
        },
    });

    // Add indexes
    listSchema.index({ owner: 1, createdAt: -1 });
    listSchema.index({ "members.user": 1 });
    ```

1.2 List Management
    ```javascript
    // list.controller.js
    export const createList = async (req, res, next) => {
        try {
            const { name, description, isPrivate } = req.body;
            const owner = req.user._id;

            const list = await List.create({
                name,
                description,
                owner,
                isPrivate,
            });

            res.status(201).json(list);
        } catch (error) {
            next(error);
        }
    };

    export const addToList = async (req, res, next) => {
        try {
            const { listId } = req.params;
            const { userId } = req.body;
            const owner = req.user._id;

            const list = await List.findOne({ _id: listId, owner });
            if (!list) {
                return next(errorHandler(404, "List not found"));
            }

            // Check if user is already in list
            if (list.members.some(member => member.user.toString() === userId)) {
                return next(errorHandler(400, "User already in list"));
            }

            list.members.push({ user: userId });
            await list.save();

            res.status(200).json(list);
        } catch (error) {
            next(error);
        }
    };
    ```

2. USER SEARCH
------------
2.1 Search Implementation
    ```javascript
    // search.controller.js
    export const searchUsers = async (req, res, next) => {
        try {
            const { query, page = 1, limit = 20 } = req.query;
            const currentUser = req.user._id;

            // Build search query
            const searchQuery = {
                $or: [
                    { username: { $regex: query, $options: "i" } },
                    { name: { $regex: query, $options: "i" } },
                    { bio: { $regex: query, $options: "i" } },
                ],
                _id: { $ne: currentUser }, // Exclude current user
            };

            // Add privacy filters
            if (!req.user) {
                searchQuery.isPrivate = false;
            }

            const users = await User.find(searchQuery)
                .select("username name profilePicture bio followersCount followingCount")
                .skip((page - 1) * limit)
                .limit(limit);

            res.status(200).json(users);
        } catch (error) {
            next(error);
        }
    };
    ```

3. USER RELATIONSHIPS
-------------------
3.1 Relationship Model
    ```javascript
    // relationship.model.js
    const relationshipSchema = new mongoose.Schema({
        user: {
            type: mongoose.Schema.Types.ObjectId,
            ref: "User",
            required: true,
        },
        target: {
            type: mongoose.Schema.Types.ObjectId,
            ref: "User",
            required: true,
        },
        type: {
            type: String,
            enum: ["follow", "block", "mute"],
            required: true,
        },
        settings: {
            // Mute settings
            hidePosts: { type: Boolean, default: false },
            hideStories: { type: Boolean, default: false },
            hideNotifications: { type: Boolean, default: false },
            // Block settings
            preventMessages: { type: Boolean, default: true },
            preventMentions: { type: Boolean, default: true },
        },
        createdAt: {
            type: Date,
            default: Date.now,
        },
    });

    // Add indexes
    relationshipSchema.index({ user: 1, target: 1, type: 1 }, { unique: true });
    ```

3.2 Relationship Management
    ```javascript
    // relationship.controller.js
    export const updateRelationship = async (req, res, next) => {
        try {
            const { targetId } = req.params;
            const { type, settings } = req.body;
            const user = req.user._id;

            // Check if target exists
            const target = await User.findById(targetId);
            if (!target) {
                return next(errorHandler(404, "User not found"));
            }

            // Update or create relationship
            const relationship = await Relationship.findOneAndUpdate(
                { user, target: targetId, type },
                { settings },
                { upsert: true, new: true }
            );

            // Update user counts if following
            if (type === "follow") {
                await User.findByIdAndUpdate(user, { $inc: { followingCount: 1 } });
                await User.findByIdAndUpdate(targetId, { $inc: { followersCount: 1 } });
            }

            res.status(200).json(relationship);
        } catch (error) {
            next(error);
        }
    };
    ```

    ```javascript
        },
    });

    // Add indexes
    messageSchema.index({ sender: 1, recipient: 1, createdAt: -1 });
    messageSchema.index({ read: 1, createdAt: -1 });
    ```

5.2 Message Management
    ```javascript
    // message.controller.js
    export const sendMessage = async (req, res, next) => {
        try {
            const { recipientId, content, media } = req.body;
            const sender = req.user._id;

            // Check if recipient exists
            const recipient = await User.findById(recipientId);
            if (!recipient) {
                return next(errorHandler(404, "User not found"));
            }

            // Check if blocked
            const isBlocked = await Relationship.exists({
                user: recipientId,
                target: sender,
                type: "block",
            });
            if (isBlocked) {
                return next(errorHandler(403, "Cannot send message to this user"));
            }

            const message = await Message.create({
                sender,
                recipient: recipientId,
                content,
                media,
            });

            // Emit real-time update
            io.to(recipientId).emit("new_message", message);

            res.status(201).json(message);
        } catch (error) {
            next(error);
        }
    };
    ```
